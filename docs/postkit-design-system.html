Swift — Dynamic Type compatible

All styles use Apple's named text styles (.headline, .title, etc.) — they auto-scale with Dynamic Type from xSmall to AX5 (Accessibility 3XL). Never hardcode .system(size:) unless you wrap with relativeTo:.

enum Typography {
    // Apple text styles — all scale with Dynamic Type automatically
    static let largeTitle   = Font.system(.largeTitle, design: .default, weight: .bold)
    static let title        = Font.system(.title, design: .default, weight: .semibold)
    static let title2       = Font.system(.title2, design: .default, weight: .semibold)
    static let title3       = Font.system(.title3, design: .default, weight: .semibold)
    static let headline     = Font.system(.headline)              // .semibold by default
    static let body         = Font.system(.body)
    static let callout      = Font.system(.callout)
    static let subheadline  = Font.system(.subheadline, design: .default, weight: .medium)
    static let footnote     = Font.system(.footnote)
    static let caption      = Font.system(.caption, design: .default, weight: .medium)
    static let caption2     = Font.system(.caption2, design: .default, weight: .medium)
    static let mono         = Font.system(.footnote, design: .monospaced).weight(.medium)

    // "Display" — bigger than largeTitle, scales relative to it
    static let display = Font.system(size: 40, weight: .bold).leading(.tight)
}

// Uppercase eyebrow helper
extension View {
    func eyebrow() -> some View {
        self.font(Typography.caption)
            .tracking(0.6)
            .textCase(.uppercase)
            .foregroundStyle(Palette.text3)
    }
}

// Optional: cap Dynamic Type for tight layouts where AX sizes break the design
extension View {
    func capDynamicType(_ max: DynamicTypeSize = .accessibility3) -> some View {
        dynamicTypeSize(...max)
    }
}

Accessibility

Six axes to support: Dynamic Type, Tap Targets, VoiceOver, Reduce Motion, Reduce Transparency, Differentiate Without Color. Every component below has hooks for all of them. Test on real device with VoiceOver before shipping.

struct PillarRowView: View {
    @Environment(\.dynamicTypeSize) var dynamicType
    let pillar: PillarSnapshot

    var body: some View {
        if dynamicType.isAccessibilitySize {
            VStack(alignment: .leading, spacing: Layout.Stack.cozy) { content }
        } else {
            HStack(spacing: Layout.Stack.comfy) { content }
        }
    }

    @ViewBuilder var content: some View { /* … */ }
}

Tap targets — 44 × 44pt minimum

HIG requirement. Any interactive element (button, chip, icon) must have a minimum 44×44pt hit area, even if its visual is smaller.

×
Visual: 24pt · Hit area: 44pt ✓
❌ Don't shrink hit area below 44pt
✓ Visual can be smaller, padding extends hit 
extension View {
    /// Ensures HIG-compliant 44pt minimum hit area (visual can be smaller)
    func adaptiveTapTarget() -> some View {
        self
            .frame(minWidth: 44, minHeight: 44)
            .contentShape(Rectangle())
    }
}

// Usage on a small icon button
Button { store.send(.closeTapped) } label: {
    Image(systemName: "xmark").font(.system(size: 14, weight: .bold))
}
.adaptiveTapTarget()

VoiceOver — labels, hints, traits

Every interactive element needs a label. Cards combine their children into a single accessibility element. Use SF Symbols (auto-described) instead of emoji where possible for buttons.

// PillarRow — combine children + describe
HStack { /* emoji + name + progress + count */ }
    .accessibilityElement(children: .combine)
    .accessibilityLabel(Text("\(pillar.name)"))
    .accessibilityValue(Text("\(pillar.photoCount) photos, \(pillar.coveragePct) percent coverage"))
    .accessibilityHint(Text("Opens pillar detail"))

struct AccessibleCardModifier: ViewModifier {
    @Environment(\.accessibilityReduceTransparency) var reduceTransparency

    func body(content: Content) -> some View {
        content
            .background(
                reduceTransparency
                    ? AnyShapeStyle(Palette.surface)
                    : AnyShapeStyle(Palette.glassStrong)
            )
            .overlay(
                RoundedRectangle(cornerRadius: Radius.card)
                    .stroke(Palette.border)
            )
            .clipShape(RoundedRectangle(cornerRadius: Radius.card))
            .shadow(color: .black.opacity(0.06), radius: 12, y: 2)
    }
}

extension View {
    func cardStyle() -> some View { modifier(AccessibleCardModifier()) }
}

Reduce Motion — replace springs with fades

Disables parallax, bounce, and spring animations for users prone to motion sickness. The Classification swipe card animation, the slot-machine reveal in onboarding, and the bento card hover effects all need fallbacks.

struct SwipeCardView: View {
    @Environment(\.accessibilityReduceMotion) var reduceMotion
    @State var offset: CGSize = .zero

    var animation: Animation {
        reduceMotion
            ? .easeInOut(duration: 0.2)
            : .interactiveSpring(response: 0.35, dampingFraction: 0.7)
    }

    var rotation: Double {
        reduceMotion ? 0 : Double(offset.width / 20)
    }

    // transitions: .opacity instead of .slide when reduceMotion
    var transition: AnyTransition {
        reduceMotion ? .opacity : .move(edge: .trailing).combined(with: .opacity)
    }
}

struct StatusDot: View {
    @Environment(\.accessibilityDifferentiateWithoutColor) var differentiate
    let status: ClassifiedPhoto.PhotoStatus

    var body: some View {
        Circle()
            .fill(color)
            .frame(width: 14, height: 14)
            .overlay(Text(symbol).font(.system(size: 8, weight: .bold)).foregroundStyle(.white))
            // when differentiate is true, also add an outline ring for extra distinction
            .overlay(differentiate ? Circle().stroke(.white, lineWidth: 1.5) : nil)
    }
}

Increase Contrast — bolder borders, darker text-3

Apple's high-contrast mode bumps border weights and uses higher-contrast text. PostKit's text-3 (#8e8e93) becomes text-2 (#3c3c43) and glass borders thicken.

extension Palette {
    static func secondaryText(_ contrast: ColorSchemeContrast) -> Color {
        contrast == .increased ? text2 : text3
    }
}

struct PillarRowView: View {
    @Environment(\.colorSchemeContrast) var contrast
    // …
    Text("\(pillar.photoCount) photos")
        .foregroundStyle(Palette.secondaryText(contrast))
}

Dynamic TypeTest at xSmall · Large · AX3 · AX5. Layout reflows, no truncation.
Tap targetsEvery interactive ≥ 44×44pt hit area (use .adaptiveTapTarget()).
VoiceOverRun with eyes closed. Every action is announceable. Cards combine children.
Reduce MotionSpring/parallax → fade. No motion sickness triggers.
Reduce TransparencyGlass → solid surface. Still legible.

extension Layout {
    enum Stack {
        static let tight:  CGFloat = 4     // label + secondary label
        static let cozy:   CGFloat = 8     // chips in a row
        static let comfy:  CGFloat = 12    // cards in a list
        static let roomy:  CGFloat = 20    // section to section
        static let loose:  CGFloat = 32    // major content breaks
    }

    enum Grid {
        static let photoGrid: CGFloat = 4      // gap between photo cells
        static let bento:     CGFloat = 8      // pillar bento gap
        static let pills:     CGFloat = 6      // horizontal scroll pills
    }
}

struct PhotoCell: View {
    let snapshot: ClassifiedPhotoSnapshot
    let image: UIImage?

    var body: some View {
        ZStack(alignment: .bottomLeading) {
            if let image {
                Image(uiImage: image)
                    .resizable().aspectRatio(1, contentMode: .fill)
            } else {
                Color(.systemGray5)
            }

            if let emoji = snapshot.pillarEmoji {
                Text("\(emoji) \(snapshot.pillarName ?? "")")
                    .font(Typography.caption2)
                    .foregroundStyle(.white)
                    .padding(Layout.Padding.chip)
                    .background(.black.opacity(0.55), in: Capsule())
                    .padding(Spacing.xxs)
            }

            StatusDot(status: snapshot.status)
                .padding(Spacing.xxs)
                .frame(maxWidth: .infinity, maxHeight: .infinity, alignment: .topTrailing)
        }
        .aspectRatio(1, contentMode: .fill)
        .clipShape(RoundedRectangle(cornerRadius: Radius.tile))
    }
}

struct StatusDot: View {
    let status: ClassifiedPhoto.PhotoStatus
    var body: some View {
        Circle().fill(color).frame(width: 14, height: 14)
            .overlay(Text(symbol).font(.system(size: 8, weight: .bold)).foregroundStyle(.white))
    }
    var color: Color {
        switch status {
        case .classified: Palette.green
        case .pending:    Palette.yellow
        case .rejected:   Palette.red
        }
    }
    var symbol: String {
        switch status { case .classified: "✓"; case .pending: "?"; case .rejected: "✗" }
    }
}

struct ProgressBar: View {

        GeometryReader { geo in

struct EmptyStateView: View {

    let title: String
    let body: String
    var actionTitle: String? = nil
    var onAction: (() -> Void)? = nil

    var body: some View {
        VStack(spacing: Layout.Stack.cozy) {
            Text(icon).font(.system(size: 40)).opacity(0.5)
            Text(title).font(Typography.headline)
            Text(body).font(Typography.footnote)

        .background(Palette.glassStrong, in: RoundedRectangle(cornerRadius: Radius.card))

struct TipCard: View {
    let icon: String
    let emphasis: String
    let message: String
    var tint: Color = Palette.purple

    var body: some View {
        HStack(spacing: Layout.Stack.cozy) {
            Text(icon).font(.system(size: 20))
            VStack(alignment: .leading, spacing: 2) {
                (Text(emphasis).bold().foregroundColor(tint) + Text(" ") + Text(message))
                    .font(Typography.callout)
            }
        }
        .padding(Layout.Padding.card)
        .background(tint.opacity(0.08), in: RoundedRectangle(cornerRadius: Radius.input))
        .overlay(RoundedRectangle(cornerRadius: Radius.input).stroke(tint.opacity(0.15)))
    }
}

struct ConfidenceBadge: View {
    let result: ClassificationResult

    var body: some View {
        HStack(spacing: Spacing.xxs) {
            Text("AI \(Int(result.confidence * 100))%")
            Image(systemName: result.confidence >= 0.70 ? "checkmark" : "cloud")
        }
        .font(Typography.caption)
        .padding(Layout.Padding.chip)
        .background(tint.opacity(0.12), in: Capsule())
        .foregroundStyle(tint)
    }
    var tint: Color { result.confidence >= 0.70 ? Palette.green : Palette.yellow }
}
